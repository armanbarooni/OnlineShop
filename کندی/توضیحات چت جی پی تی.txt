$requestStartById = @{}
$requests = @()
$syncRuns = @()
$outgoingRuns = @()
$syncStart = $null
$outgoingStart = $null

Get-Content $LogPath | ForEach-Object {
    $line = $_

    if ($line -match '^(?<ts>\d{4}-\d{2}-\d{2} \d{2}:\d{2}:\d{2}\.\d{3} [+-]\d{2}:\d{2})') {
        $timestamp = [DateTimeOffset]::ParseExact($matches.ts, 'yyyy-MM-dd HH:mm:ss.fff zzz', [Globalization.CultureInfo]::InvariantCulture)

        if ($line -like '*Starting Mahak Sync...*') {
            $syncStart = $timestamp
        } elseif ($line -like '*Mahak Sync Completed Successfully.*' -and $syncStart) {
            $syncRuns += [pscustomobject]@{
                Start = $syncStart
                End = $timestamp
                DurationMs = [math]::Round(($timestamp - $syncStart).TotalMilliseconds)
            }
            $syncStart = $null
        } elseif ($line -like '*Starting outgoing sync to Mahak...*') {
            $outgoingStart = $timestamp
        } elseif (($line -like '*No orders to sync to Mahak*' -or $line -like '*outgoing sync completed*' -or $line -like '*orders synced to Mahak*') -and $outgoingStart) {
            $outgoingRuns += [pscustomobject]@{
                Start = $outgoingStart
                End = $timestamp
                DurationMs = [math]::Round(($timestamp - $outgoingStart).TotalMilliseconds)
            }
            $outgoingStart = $null
        }
    }

    if ($line -match 'Request (?<id>[0-9a-f]+): (?<method>\S+) (?<path>\S+) ') {
        $requestStartById[$matches.id] = [pscustomobject]@{
            Method = $matches.method
            Path = $matches.path
        }
    } elseif ($line -match 'Request (?<id>[0-9a-f]+) completed: (?<code>\d{3}) in (?<ms>\d+)ms') {
        $id = $matches.id
        if ($requestStartById.ContainsKey($id)) {
            $meta = $requestStartById[$id]
            $requests += [pscustomobject]@{
                Method = $meta.Method
                Path = $meta.Path
                Code = [int]$matches.code
                Ms = [int]$matches.ms
            }
            $requestStartById.Remove($id) | Out-Null
        }
    }
}

Write-Host "=== Request Summary ==="
if ($requests.Count -gt 0) {
    $requests | Measure-Object Ms -Average -Maximum -Minimum | Format-List

    Write-Host "`nTop $Top Slow Requests"
    $requests | Sort-Object Ms -Descending | Select-Object -First $Top Method, Path, Code, Ms | Format-Table -AutoSize
